    16.1. تثبيت SGBD و MySQL

    للحصول على SGBD و MySQL، سنقوم بتثبيت برنامج Laragon.

    16.1.1. تثبيت برنامج Laragon

    • لغة البرمجة النصية PHP التي لن نستخدمها؛

    16.2. تثبيت الحزمة [mysql-connector-python]

    سنقوم بكتابة نصوص برمجية بلغة Python لاستغلال قاعدة البيانات التي تم إنشاؤها مسبقًا بالبنية التالية:

    يُستخدم الموصل لعزل كود Python عن جهاز SGBD قيد التشغيل. توجد موصلات لمختلف أجهزة SGBD، وتلتزم جميعها بنفس الواجهة. لذلك، عندما يتم استبدال SGBD وMySQL بـ SGBD وPostgreSQL في المثال أعلاه، تصبح البنية كما يلي:

    نظرًا لأن جميع موصلات SGBD تتوافق جميعها مع نفس الواجهة، فلا داعي عادةً لتعديل البرنامج النصي Python. وفي الواقع، فإن معظم موصلات SGBD لها موصل SQL خاص بها:

    • وهي تتوافق مع معيار SQL (لغة الاستعلام الهيكلية)؛
    • ولكنها توسع نطاقها، نظرًا لعدم كفايتها، بإضافات خاصة باللغة؛

    لذلك، غالبًا ما يتطلب تغيير قاعدة البيانات SGBD إجراء تعديلات على النصوص البرمجية المتعلقة بـ SQL.

    لا توفر لغة Python أصلاً إمكانية إدارة قاعدة بيانات MySQL. ولذلك، يجب تنزيل حزمة برمجية. وهناك العديد من الحزم المتاحة. سنستخدم هنا حزمة [mysql-connector-python] التي تُعد الموصل الرسمي لشركة Oracle، وهي الشركة المالكة لـ MySQL.

# استيراد الوحدة النمطية mysql.connector
from mysql.connector import DatabaseError, InterfaceError, connect

def connexion(host: str, database: str, login: str, pwd: str):
    # يُجري الاتصال ثم يقطع الاتصال (اسم المستخدم، كلمة المرور) بقاعدة البيانات [database] على الخادم [host]
    # يطلق الاستثناء DatabaseError في حالة وجود مشكلة
    connexion = None

        # الاتصال

        print(
            f"Connexion réussie à la base database={database}, host={host} sous l'identité user={login}, passwd={pwd}")
    finally:
        # يتم إغلاق الاتصال إذا كان مفتوحًا
        if connexion:
            connexion.close()
            print("Déconnexion réussie\n")

    16.5. نص برمجي [mysql_03]: إنشاء جدول MySQL

    الآن بعد أن تعلمنا كيفية إنشاء اتصال باستخدام SGBD و MySQL، سنبدأ في إصدار أوامر SQL عبر هذا الاتصال. للقيام بذلك، سنقوم بالاتصال بقاعدة البيانات التي تم إنشاؤها [dbpersonnes] واستخدام هذا الاتصال لإنشاء جدول داخل قاعدة البيانات.

    
# عمليات الاستيراد
import sys

from mysql.connector import DatabaseError, InterfaceError, connect
from mysql.connector.connection import MySQLConnection


# ---------------------------------------------------------------------------------
def execute_sql(connexion: MySQLConnection, update: str):
    # تنفيذ استعلام تحديث على عملية الاتصال
    curseur = None
    try:
        # يتم طلب مؤشر
        curseur = connexion.cursor()

def execute_list_of_commands(connexion: MySQLConnection, sql_commands: list,
                             suivi: bool = False, arrêt: bool = True, with_transaction: bool = True):
    # استخدام الاتصال المفتوح [connexion]
    # تنفيذ الأوامر SQL الواردة في القائمة [sql_commands] عبر هذا الاتصال
    # هذا الملف هو ملف أوامر SQL يتم تنفيذها بمعدل أمر واحد في كل سطر
    # إذا كانت قيمة «المتابعة» = «صحيح»، فإن كل عملية تنفيذ لأمر SQL تُعرض مع إشارة إلى نجاحها أو فشلها
    # إذا كانت «إيقاف» = «صحيح»، تتوقف الوظيفة عند أول خطأ تصادفه، وإلا فإنها تنفذ جميع أوامر SQL
    # إذا كان with_transaction=True، فإن أي خطأ يلغي جميع الأوامر SQL التي تم تنفيذها مسبقًا
    # إذا كانت قيمة with_transaction تساوي False، فإن أي خطأ لا يؤثر على الأوامر SQL التي تم تنفيذها مسبقًا
    # تُرجع الدالة قائمة [erreur1, erreur2, ...]

    curseur = None
    connexion.autocommit = not with_transaction
    erreurs = []
    try:
        # مطلوب شريط تمرير
        curseur = connexion.cursor()
        # تنفيذ sql_commands SQL الموجودة في sql_commands
        # يتم تنفيذها واحدة تلو الأخرى
        for command in sql_commands:
            # يتم حذف المسافات في بداية ونهاية الأمر الحالي
            command = command.strip()
            # هل لدينا أمر فارغ أم تعليق؟ إذا كان الجواب نعم، ننتقل إلى الأمر التالي
            if command == '' or command[0] == "#":
                continue
            # تنفيذ الأمر الحالي
            error = None
            try:
                curseur.execute(command)
            except (InterfaceError, DatabaseError) as erreur:
                error = erreur
            # هل حدث خطأ؟
            if error:
                # خطأ آخر
                msg = f"{command} : Erreur ({error})"
                erreurs.append(msg)

                    print(msg)

            else:
                # التصديق

# عمليات الاستيراد
from mysql.connector import connect, DatabaseError, InterfaceError

# هوية المستخدم
ID = "admpersonnes"
PWD = "nobody"
# الجهاز المضيف لنظام إدارة قواعد البيانات
HOST = "localhost"
# هوية قاعدة البيانات
BASE = "dbpersonnes"

# قائمة الأشخاص (الاسم، الاسم الأول، العمر)
personnes = []
for i in range(5):
    personnes.append((i, f"n0{i}", f"p0{i}", i + 10))
personnes.append((40, "d'Aboot", "Y'éna", 18))
# قائمة أخرى بالأشخاص
autresPersonnes = []
for i in range(5):
    autresPersonnes.append((i + 100, f"n1{i}", f"p1{i}", i + 20))
autresPersonnes.append((200, "d'Aboot", "F'ilhem", 34))

# الوصول إلى SGBD
connexion = None
try:
    # تسجيل الدخول
    connexion = connect(host=HOST, user=ID, password=PWD, database=BASE)
    # مؤشر
    curseur = connexion.cursor()
    # حذف السجلات الموجودة
    curseur.execute("delete from personnes")
    # الإدراج شخصًا تلو الآخر باستخدام استعلام مُعد مسبقًا
    for personne in personnes:
        curseur.execute("insert into personnes(id,nom,prenom,age) values(%s,%s,%s,%s)", personne)
    # إدراج قائمة بالأشخاص دفعة واحدة
    curseur.executemany("insert into personnes(id,nom,prenom,age) values(%s, %s,%s,%s)", autresPersonnes)
    # التحقق من صحة المعاملة
    connexion.commit()
except (DatabaseError, InterfaceError) as erreur:
    # عرض الخطأ
    print(f"L'erreur suivante s'est produite : {erreur}")
    # إلغاء المعاملة
    if connexion:
        connexion.rollback()
finally:
    # إنهاء الاتصال
    if connexion:
        connexion.close()

    تكمن فائدة الاستعلامات المعلمة في نقطتين:

    • يتم تنفيذها بشكل أسرع من الاستعلامات «الثابتة» التي يتعين تحليلها عند كل عملية تنفيذ. لا يتم تحليل الاستعلام المعلم [executemany] إلا مرة واحدة. بعد ذلك يتم تنفيذه n مرات دون الحاجة إلى تحليله مرة أخرى؛
    • يتم التحقق من المعلمات التي يتم إدخالها في الاستعلام المعلم. إذا كانت تحتوي على أحرف محجوزة، مثل الفاصلة العليا على سبيل المثال، يتم «حمايتها» حتى لا تتداخل مع تنفيذ الأمر SQL. وللتحقق من هذه النقطة، تم تضمين أسماء وألقاب تحتوي على علامات الفاصلة العليا في القائمة (السطران 16 و21)؛
